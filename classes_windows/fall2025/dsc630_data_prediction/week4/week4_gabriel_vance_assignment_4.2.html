# Compare clustering performance between original and cleaned data
df_original = df  # Original with basic drops
df_cleaned = df.drop(drop_cols , axis=1)  # Cleaned with additional drops

# Scale both datasets
scaler_orig = StandardScaler()
scaled_data_orig = scaler_orig.fit_transform(df_original)

scaler_clean = StandardScaler()
scaled_data_clean = scaler_clean.fit_transform(df_cleaned)

# Calculate silhouette scores for both datasets
silhouette_scores_orig = []
silhouette_scores_clean = []
cluster_range = range(2, 15)

for n_clusters in cluster_range:
    # Original data
    kmeans_orig = KMeans(n_clusters=n_clusters, random_state=42)
    cluster_labels_orig = kmeans_orig.fit_predict(scaled_data_orig)
    score_orig = silhouette_score(scaled_data_orig, cluster_labels_orig)
    silhouette_scores_orig.append(score_orig)
    
    # Cleaned data
    kmeans_clean = KMeans(n_clusters=n_clusters, random_state=42)
    cluster_labels_clean = kmeans_clean.fit_predict(scaled_data_clean)
    score_clean = silhouette_score(scaled_data_clean, cluster_labels_clean)
    silhouette_scores_clean.append(score_clean)

# Plot both silhouette scores on the same chart
plt.figure(figsize=(10, 6))
plt.plot(cluster_range, silhouette_scores_orig, marker='o', label='Original Data', linewidth=2)
plt.plot(cluster_range, silhouette_scores_clean, marker='s', label='Cleaned Data', linewidth=2)
plt.xlabel('Number of Clusters')
plt.ylabel('Silhouette Score')
plt.title('Silhouette Score Comparison: Original vs Cleaned Data')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

# Print max scores for comparison
max_orig = max(silhouette_scores_orig)
max_clean = max(silhouette_scores_clean)
print(f"Max silhouette score (Original): {max_orig:.3f}")

print(f"Improvement: {max_clean - max_orig:.3f}")

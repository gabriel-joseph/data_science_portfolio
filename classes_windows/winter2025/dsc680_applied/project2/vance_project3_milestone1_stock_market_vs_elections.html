Project 2 Milestone 3: Stock Market Performance vs Elections¶

This notebook follows the problem statement in Vance Project 2 Milestone 3.pdf and answers:

  1. Are future S&P 500 returns statistically similar under both major political parties?
  2. Are incumbent political parties more likely to lose elections during periods of S&P 500 underperformance?
  3. Given the current political landscape, what is the predicted stock market return from January 2026 to January 2027?

The workflow caches downloaded market data to local CSV files so data is not re-downloaded each run.

Approach¶

  • Use yfinance for historical S&P 500 data (^GSPC) and cache it to data/sp500_daily.csv.
  • Compute January-to-January returns as the annual performance measure.
  • Add U.S. presidential party labels by year and election outcomes for presidential election years.
  • Apply statistical tests, visual analysis, and a simple regression forecast model.

Note: This is a historical association analysis, not causal proof or trading advice.

def load_sp500(cache_path=DATA_DIR / "sp500_daily.csv", start="1950-01-01"):
    """Load cached S&P 500 daily data or download and cache it."""
    if cache_path.exists():
        df = pd.read_csv(cache_path, parse_dates=["Date"], index_col="Date")
        source = "cache"
    else:
        df = yf.download("^GSPC", start=start, progress=False, auto_adjust=True)
        if isinstance(df.columns, pd.MultiIndex):
            df.columns = [c[0] for c in df.columns]
        keep_cols = [
            c for c in ["Open", "High", "Low", "Close", "Volume"] if c in df.columns
        ]
        df = df[keep_cols].dropna(subset=["Close"])
        df.to_csv(cache_path, index_label="Date")
        source = "download"

    if "Close" not in df.columns:
        raise ValueError("Expected a Close column in S&P 500 dataset.")

    return df.sort_index(), source

# Build January-to-January annual return series
jan_prices = sp500_daily[sp500_daily.index.month == 1].copy()
jan_prices["year"] = jan_prices.index.year
jan_first_close = jan_prices.groupby("year")["Close"].first()

annual_returns = (
    (jan_first_close.shift(-1) / jan_first_close - 1)
    .dropna()
    .to_frame("return_jan_to_jan")
)
annual_returns.index.name = "year"
annual_returns["return_pct"] = annual_returns["return_jan_to_jan"] * 100
annual_returns["is_presidential_election_year"] = (
    annual_returns.index % 4 == 0
).astype(int)
annual_returns["is_midterm_year"] = (annual_returns.index % 4 == 2).astype(int)
annual_returns["is_even_year"] = (annual_returns.index % 2 == 0).astype(int)

# Presidential party mapping by year (party in power for most of the Jan-to-Jan window)
PARTY_PERIODS = [
    (1950, 1952, "Democrat"),
    (1953, 1960, "Republican"),
    (1961, 1968, "Democrat"),
    (1969, 1976, "Republican"),
    (1977, 1980, "Democrat"),
    (1981, 1992, "Republican"),
    (1993, 2000, "Democrat"),
    (2001, 2008, "Republican"),
    (2009, 2016, "Democrat"),
    (2017, 2020, "Republican"),
    (2021, 2024, "Democrat"),
    (2025, 2026, "Republican"),
]


def get_party(year):
    for start, end, party in PARTY_PERIODS:
        if start <= year <= end:
            return party
    return np.nan


annual_returns["president_party"] = annual_returns.index.map(get_party)

annual_returns.to_csv(DATA_DIR / "sp500_annual_returns_jan_to_jan.csv", index=True)

dem = annual_returns.loc[
    annual_returns["president_party"] == "Democrat", "return_jan_to_jan"
].dropna()
rep = annual_returns.loc[
    annual_returns["president_party"] == "Republican", "return_jan_to_jan"
].dropna()

# Welch's t-test and Mann-Whitney U (robust non-parametric alternative)
test_t = stats.ttest_ind(dem, rep, equal_var=False, nan_policy="omit")
test_u = stats.mannwhitneyu(dem, rep, alternative="two-sided")

# Cohen's d effect size
pooled_sd = np.sqrt(((dem.var(ddof=1) + rep.var(ddof=1)) / 2))
cohens_d = (dem.mean() - rep.mean()) / pooled_sd

# Bootstrap confidence interval for mean difference (Dem - Rep)
rng = np.random.default_rng(680)
boot = []
for _ in range(5000):
    boot_dem = rng.choice(dem.to_numpy(), size=len(dem), replace=True).mean()
    boot_rep = rng.choice(rep.to_numpy(), size=len(rep), replace=True).mean()
    boot.append(boot_dem - boot_rep)
boot = np.array(boot)
ci_low, ci_high = np.percentile(boot, [2.5, 97.5])

q1_stats = pd.DataFrame(
    {
        "Metric": [
            "Democratic mean return",

# Underperformance measured as compounded return over the two years before election year
def get_pre2_return(election_year):
    y1, y2 = election_year - 2, election_year - 1
    if y1 in ret.index and y2 in ret.index:
        return (1 + ret.loc[y1]) * (1 + ret.loc[y2]) - 1
    return np.nan


elections["pre2y_return"] = elections["election_year"].map(get_pre2_return)

X = model_df[feature_cols]

# Rolling-origin time series CV for realistic error estimate
cv = TimeSeriesSplit(n_splits=8)
cv_pred = pd.Series(index=model_df.index, dtype=float)
for train_idx, test_idx in cv.split(X):
    m = LinearRegression()
    m.fit(X.iloc[train_idx], y.iloc[train_idx])
    cv_pred.iloc[test_idx] = m.predict(X.iloc[test_idx])

mask = cv_pred.notna()
cv_mae = mean_absolute_error(y[mask], cv_pred[mask])
cv_rmse = np.sqrt(mean_squared_error(y[mask], cv_pred[mask]))
cv_r2 = 1 - np.sum((y[mask] - cv_pred[mask]) ** 2) / np.sum(
    (y[mask] - y[mask].mean()) ** 2
)
resid_std = (y[mask] - cv_pred[mask]).std(ddof=1)

# Fit final model on full training history
final_model = LinearRegression()
final_model.fit(X, y)
coef_series = pd.Series(final_model.coef_, index=feature_cols).sort_values()

Interpretation: the model suggests an expected negative direction based on current inputs, but uncertainty dominates. This is best treated as a scenario estimate, not a precise forecast.
